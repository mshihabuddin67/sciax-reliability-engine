async function analyze() { const inputText = document.getElementById("textInput").value; const res = await fetch("https://sciax-reliability-engine.onrender.com/analyze", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ text: inputText }) }); const data = await res.json(); console.log(data); document.getElementById("output").innerText = JSON.stringify(data, null, 2); }
