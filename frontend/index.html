async function analyze(){ const text = document.getElementById("textInput").value; const resultBox = document.getElementById("resultBox"); try{ const res = await fetch( "https://sciax-reliability-engine.onrender.com/analyze", { method:"POST", headers:{ "Content-Type":"application/json", "x-api-key":"sciax-demo-key-123" }, body:JSON.stringify({ text:text }) } ); const data = await res.json(); console.log("API RESPONSE:", data); resultBox.style.display = "block"; const analysis = data.analysis; const riskText = document.getElementById("riskText"); riskText.innerText = "Risk Level: " + analysis.risk_level; riskText.className = analysis.risk_level.toLowerCase(); document.getElementById("stability").innerText = analysis.stability_score; document.getElementById("conflict").innerText = analysis.conflict_score || "N/A"; document.getElementById("reason").innerText = data.explainability ? data.explainability.join(", ") : "No reason provided"; } catch(error){ console.error(error); alert("API Error"); } }
